from pathlib import Path
import collections,hashlib,json,re,subprocess,sys
original=Path(sys.argv[1])/'AURA_ANDROID_REL_04_R13_INDEPENDENT_EVIDENCE_39252'
root=Path('docs/android/evidence')
prior,current=root/'rel04-r13-independent-39252',root/'rel04-r14'
impl='0df08780bd75e16b676efda6c99c1738bc72681b'
h=Path('docs/android/AURA_ANDROID_REL_04_R14_VERIFICATION_HANDOVER.md')
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
git=lambda *a:subprocess.check_output(['git',*a])
entries=re.findall(r'^\| .([^\x60]+). \| .([0-9a-f]{64}). \|$',h.read_text(),re.M)
assert len(entries)==16
for name,digest in entries:
    assert sha(Path(name))==digest,name
    print(digest,name)
lines=(prior/'SHA256SUMS').read_text().splitlines()
assert len(lines)==50
for l in lines:
    digest,n=l.split()
    path=prior/n
    if not path.exists(): path=prior/(n+'.txt')
    assert sha(path)==digest and path.read_bytes()==(original/n).read_bytes(),n
print('50 prior independent files verified against original bytes')
for n in ('offlineHabitQueue','activitySync','signOut'):
    assert git('show','a8a9767:apps/web/lib/'+n+'.ts')==Path('test/fixtures/rel04-r13-head',n+'.ts.txt').read_bytes(),n
    print('Exact R13 fixture verified',n)
for folder,n in [('math-core-tests',255),('db-and-browser-tests',78)]:
    p=current/folder/'manifest.json'
    m=json.loads(p.read_text())
    assert m['head']==impl and m['codeTreeClean'] and len(m['commands'])==n
    assert {x['log'] for x in m['commands']}=={str(x.relative_to(p.parent)) for x in (p.parent/'logs').glob('*.log')}
    for c in m['commands']:
        assert sha(p.parent/c['log'])==c['logSha256'],c['log']
    print(folder,n,dict(collections.Counter((c['job'],c['exit']) for c in m['commands'])))
