const { chromium } = require(require.resolve('@playwright/test', { paths: [root] }));
const rev = 'd09449b';
const source = file => execFileSync('git', ['show', rev + ':' + file], { encoding: 'utf8' });
const code = ts.transpileModule(source('apps/web/lib/offlineHabitQueue.ts'), { compilerOptions: { module: ts.ModuleKind.CommonJS, target: ts.ScriptTarget.ES2020 } }).outputText;
const worker = source('apps/web/public/sw.js');
const legacy = '<!doctype html><script>var exports={};\n' + code + '\nwindow.legacyEnqueue=exports.enqueueOwned;</script>';
const safe = '<!doctype html><title>Queue writer excluded</title>';
let exposeLegacy = true;
function makeOrigin(withLegacy) {
  return http.createServer((req, res) => {
    if (req.url === '/sw.js' && withLegacy) { res.writeHead(200, { 'content-type': 'application/javascript' }); res.end(worker); return; }
    if (req.url !== '/') { res.writeHead(200); res.end('{}'); return; }
    res.writeHead(200, { 'content-type': 'text/html', 'cache-control': 'no-store' });
    res.end(withLegacy && exposeLegacy ? legacy : safe);
  });
}
async function endpoint(server) {
  await new Promise(resolve => server.listen(0, '127.0.0.1', resolve));
  return 'http://127.0.0.1:' + server.address().port;
}
(async () => {

  const old = await endpoint(oldServer), isolated = await endpoint(isolatedServer);
  const profile = fs.mkdtempSync(path.join(os.tmpdir(), 'r12-origins-'));
  let context;
  const key = 'offline_habit_queue_v2:preview-a';

  const inspect = page => page.evaluate(k => JSON.parse(localStorage.getItem(k) ?? '[]').map(x => x.clientRequestId), key);
  try {
    context = await chromium.launchPersistentContext(profile, { headless: true });
    const retained = await context.newPage();
    await retained.goto(old);
    await retained.evaluate(async () => {
      await navigator.serviceWorker.register('/sw.js');
      await navigator.serviceWorker.ready;

  }
