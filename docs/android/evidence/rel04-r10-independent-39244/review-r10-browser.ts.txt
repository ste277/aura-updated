// ---------------------------------------------------------------- server
type Gate = { path: string; phase: 'before' | 'after'; release: () => void; promise: Promise<void>; arrived: Promise<void>; mark: () => void };
const gates: Gate[] = [];
function hold(p: string, phase: 'before' | 'after') {
  let release!: () => void, mark!: () => void;
  const g: Gate = { path: p, phase, release: () => release(), promise: new Promise((r) => { release = r; }), arrived: new Promise((r) => { mark = r; }), mark: () => mark() };
  gates.push(g); return g;
}
function takeGate(p: string, phase: 'before' | 'after') { const i = gates.findIndex((g) => g.path === p && g.phase === phase); return i >= 0 ? gates.splice(i, 1)[0] : null; }
const serverLog: Array<{ path: string; method: string; status: number; expected: string | null }> = [];

const server = http.createServer(async (req, res) => {
  const url = new URL(req.url ?? '/', 'http://127.0.0.1');
  if (url.pathname === '/sw.js') { res.writeHead(200, { 'content-type': 'text/javascript', 'cache-control': 'no-cache' }); res.end(deploy.sw); return; }
  if (url.pathname === '/app') { res.writeHead(200, { 'content-type': 'text/html', 'cache-control': 'no-cache' }); res.end(deploy.appHtml === 'old' ? OLD_PAGE : '<!doctype html><html><body><script>navigator.serviceWorker.register("/sw.js");</script><script src="/bundle.js"></script></body></html>'); return; }
  if (url.pathname === '/') { res.writeHead(200, { 'content-type': 'text/html' }); res.end('<!doctype html><html><body><script src="/bundle.js"></script></body></html>'); return; }
  if (url.pathname === '/bundle.js') { res.writeHead(200, { 'content-type': 'text/javascript' }); res.end(BUNDLE); return; }
  const chunks: Buffer[] = [];
  for await (const c of req) chunks.push(c as Buffer);
  const body = chunks.length ? Buffer.concat(chunks).toString('utf8') : undefined;
  const headers = new Headers();
  for (const [k, v] of Object.entries(req.headers)) if (typeof v === 'string') headers.set(k, v);
  const method = (req.method ?? 'GET').toUpperCase();
  const key = `${method} ${url.pathname}`;
  const before = takeGate(url.pathname, 'before');
  if (before) { before.mark(); await before.promise; }
  const nreq = new NextRequest(url.toString(), { method, headers, body: method === 'GET' || method === 'HEAD' ? undefined : body });
  let out: Response;
  const mw: Response = url.pathname.startsWith('/api/') ? await middlewareMod.middleware(nreq) : new Response(null, { headers: { 'x-middleware-next': '1' } });
  if (mw.headers.get('x-middleware-next') !== '1') out = mw;
  else if (key === 'POST /api/auth/verify-code') {
    const uid = EMAIL[JSON.parse(body ?? '{}').email];
    out = uid
      ? Response.json({ ok: true, userId: uid }, { status: 200, headers: { 'set-cookie': `as_session=${auth.createSessionToken(uid, store.users.get(uid)!.email)}; Path=/; HttpOnly; SameSite=Lax; Max-Age=2592000`, 'Cache-Control': 'no-store' } })
      : Response.json({ error: 'That code is not valid.' }, { status: 401 });
  } else if (key === 'POST /api/auth/logout-r2') out = await r2Logout.POST(nreq);
  else if (routes[key]) out = await routes[key](nreq);
  else out = Response.json({}, { status: 200 });
  serverLog.push({ path: url.pathname, method, status: out.status, expected: headers.get('x-aura-expected-account') });
  const after = takeGate(url.pathname, 'after');
  if (after) { after.mark(); await after.promise; }
  const outHeaders: Record<string, string | string[]> = {};
  out.headers.forEach((v, k) => { if (k !== 'set-cookie') outHeaders[k] = v; });
  const sc = (out.headers as any).getSetCookie?.() ?? [];
  if (sc.length) outHeaders['set-cookie'] = sc;
  res.writeHead(out.status, outHeaders);
  res.end(Buffer.from(await out.arrayBuffer()));
});

async function main() {

  const { payload } = require('./rel04AccountIsolationHarness');
  const { chromium } = require('@playwright/test');
  BUNDLE = bundle();
  await new Promise<void>((resolve) => server.listen(0, '127.0.0.1', resolve));
  const origin = 'http://127.0.0.1:' + (server.address() as any).port;
  const profiles = fs.mkdtempSync(path.join(require('node:os').tmpdir(), 'r10-browser-'));
  const profile = path.join(profiles, 'existing');
  const key = 'offline_habit_queue_v2:' + A;
  const raw = '[{"clientRequestId":"historical"';
  let context: any;
  const launch = async (dir: string) => chromium.launchPersistentContext(dir, { headless: true });
  async function open(email: string) {
    const p = await context.newPage();
    await p.goto(origin + '/');
    await p.waitForFunction(() => (window as any).auraReady);
    const result = await p.evaluate(async (e: string) => (window as any).aura.si.verifySignInCode({ email: e, code: '123456' }), email);
    assert.equal(result.kind, 'VERIFIED');
    return p;
  }
  const accountA = 'alicesynth@example.invalid';
  const accountB = 'bobsynth@example.invalid';
  const run = (p: any, id: string) => p.evaluate(async (value: any) => {
    const w = (window as any).aura;
    const sent = await w.sync.submitActivityLog(value, 'temp');
    const rp = await w.sync.replayOfflineQueue();
    return { submit: sent.kind, replayStoppedBecause: rp.stoppedBecause, attempted: rp.attempted };
  }, payload('review browser activity', id));
  const signout = (p: any) => p.evaluate(async () => {
    const so = (window as any).aura.so;
    const r = await so.performSignOut({ fetch: window.fetch.bind(window), confirm: () => true });
    return { outcome: r.outcome, owned: r.owned, message: so.signOutOutcomeMessage(r) };
  });
  try {
    context = await launch(profile);
    const a = await open(accountA);
    const browserVersion = context.browser()?.version();
    await a.evaluate(async (value: any) => (window as any).aura.q.enqueueOwned(value), { ...payload('queued before damage', 'browser-preexisting'), ownerUserId: A, tempId: 'pre' });
    // Exercise replay before sign-out has given the pre-existing request a decision.
    await a.evaluate(([k,v]: string[]) => localStorage.setItem(k,v), [key,raw]);
    const start = await run(a,'browser-before-signout');
    const logout = await signout(a);
    const b = await open(accountB);
    const other = await run(b,'browser-b');
    await signout(b);
    const same = await open(accountA);
    const signedBackIn = await run(same,'browser-after-signin');
    await context.close();
    context = undefined;
    context = await launch(profile);
    const newA = await open(accountA);
    const restarted = await run(newA,'browser-after-restart');
    const survivor = await newA.evaluate((k: string) => localStorage.getItem(k), key);
    await context.close();
    context = undefined;
    context = await launch(path.join(profiles, 'clean'));
    const clean = await open(accountA);
    const otherProfile = await run(clean,'browser-clean-profile');
    assert.equal(survivor, raw);
    for (const x of [start,signedBackIn,restarted]) { assert.equal(x.submit,'storage-unavailable'); assert.equal(x.attempted,0); }
    assert.equal(logout.outcome,'signed-out'); assert.equal(logout.owned.unidentified,1);
    assert.equal(other.submit,'confirmed'); assert.equal(otherProfile.submit,'confirmed');
    const writes = serverLog.filter((r) => r.method === 'POST' && r.path === '/api/habit-logs');
